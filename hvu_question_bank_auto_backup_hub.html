Đoạn mã này giúp tiện ích AUTO A đọc đúng khung màu xanh ở góc trái dưới màn hình thi: 

// Tự động nhận diện Tuần ("KIỂM TRA TUẦN 9") & Tên Môn ("Quản lý HTMT") ở góc trái dưới cùng
(function initHvuAutoSubjectWeekBackup() {
    const channel = new BroadcastChannel('HVU_AUTO_A_BACKUP_CHANNEL');

    function detectSubjectAndWeekFromBottomLeft() {
        let detectedWeek = "Tuần 1";
        let detectedSubject = "";

        const elements = Array.from(document.querySelectorAll('div, span, p, footer'));
        for (const el of elements) {
            const rect = el.getBoundingClientRect();
            // Quét vùng góc trái dưới cùng màn hình
            if (rect.top > window.innerHeight * 0.70 && rect.left < window.innerWidth * 0.35 && rect.height > 15 && rect.height < 150) {
                const lines = (el.innerText || "").split('\n').map(l => l.trim()).filter(Boolean);
                for (let i = 0; i < lines.length; i++) {
                    const m = lines[i].match(/(?:KIỂM\s*TRA\s*)?TUẦN\s*(\d+)/i);
                    if (m) {
                        detectedWeek = `Tuần ${parseInt(m[1], 10)}`;
                        if (lines[i + 1] && !lines[i + 1].match(/TUẦN\s*\d+/i)) {
                            detectedSubject = lines[i + 1];
                        }
                        break;
                    }
                }
            }
            if (detectedSubject) break;
        }

        // Nếu môn chưa có tên thì tự tạo tên môn mới
        if (!detectedSubject) {
            const hash = new URLSearchParams(location.search).get('hashcode') || '';
            detectedSubject = hash ? `Môn HVU (${hash.slice(0, 5).toUpperCase()})` : `Môn mới (${new Date().toLocaleDateString('vi-VN')})`;
        }
        return { subject: detectedSubject, week: detectedWeek };
    }

    function syncCurrentQuestionToWeb() {
        const { subject, week } = detectSubjectAndWeekFromBottomLeft();
        const qNumMatch = (document.body.innerText || "").match(/Câu\s*(\d+)\s*:/i);
        const questionNumber = qNumMatch ? `Câu ${qNumMatch[1]}` : "Câu hỏi";

        const qNode = document.querySelector('.question-content, .exam-question, [class*="question"]');
        const questionText = qNode ? qNode.innerText.replace(/^Câu\s*\d+\s*:\s*/i, '').trim() : "";
        if (!questionText) return;

        const options = { A: "", B: "", C: "", D: "" };
        let correctAnswer = "A";

        document.querySelectorAll('label, .answer-item, [class*="answer"]').forEach(node => {
            const txt = (node.innerText || "").trim();
            ['A', 'B', 'C', 'D'].forEach(letter => {
                if (txt.startsWith(letter + ' ') || txt.startsWith(letter + '.')) {
                    options[letter] = txt.replace(new RegExp(`^${letter}[\\s.]+`), '').trim();
                }
                if (node.querySelector('input:checked') && txt.startsWith(letter)) {
                    correctAnswer = letter;
                }
            });
        });

        const message = {
            type: 'HVU_EXTENSION_BACKUP',
            payload: { subject, week, questionNumber, question: questionText, options, correctAnswer }
        };
        channel.postMessage(message);
        window.postMessage(message, '*');
    }

    document.addEventListener('change', () => setTimeout(syncCurrentQuestionToWeb, 300));
    document.addEventListener('click', () => setTimeout(syncCurrentQuestionToWeb, 500));
})();
